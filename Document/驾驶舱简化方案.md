# 驾驶舱简化方案（仅农户和超级管理员）

## 一、核心数据展示（4个统计卡片）

### 1. 批次统计
- **显示内容**：批次总数
- **数据来源**：`batch` 表
- **SQL查询**：
```sql
-- 超级管理员
SELECT COUNT(*) FROM batch WHERE status != -1;

-- 农户
SELECT COUNT(*) FROM batch WHERE farmer_id = ? AND status != -1;
```

### 2. 商品统计
- **显示内容**：商品总数
- **数据来源**：`product` 表
- **SQL查询**：
```sql
-- 超级管理员
SELECT COUNT(*) FROM product WHERE status = 1;

-- 农户
SELECT COUNT(*) FROM product WHERE farmer_id = ? AND status = 1;
```

### 3. 订单统计
- **显示内容**：订单总数
- **数据来源**：`order` 表
- **SQL查询**：
```sql
-- 超级管理员
SELECT COUNT(*) FROM `order`;

-- 农户
SELECT COUNT(*) FROM `order` WHERE farmer_id = ?;
```

### 4. 销售额统计
- **显示内容**：总销售额
- **数据来源**：`order` 表
- **SQL查询**：
```sql
-- 超级管理员
SELECT COALESCE(SUM(total_price), 0) FROM `order` WHERE order_status IN (3, 4);

-- 农户
SELECT COALESCE(SUM(total_price), 0) FROM `order` 
WHERE farmer_id = ? AND order_status IN (3, 4);
```

## 二、批次状态分布（饼图）

### 显示内容
- 种植中批次数量
- 已收获批次数量
- 已售罄批次数量

### 数据来源
`batch` 表的 `status` 字段

### SQL查询
```sql
-- 超级管理员
SELECT 
  CASE status
    WHEN 1 THEN '种植中'
    WHEN 2 THEN '已收获'
    WHEN 3 THEN '已售罄'
  END as name,
  COUNT(*) as value
FROM batch
WHERE status IN (1, 2, 3)
GROUP BY status;

-- 农户
SELECT 
  CASE status
    WHEN 1 THEN '种植中'
    WHEN 2 THEN '已收获'
    WHEN 3 THEN '已售罄'
  END as name,
  COUNT(*) as value
FROM batch
WHERE farmer_id = ? AND status IN (1, 2, 3)
GROUP BY status;
```

## 三、月度批次趋势（折线图）

### 显示内容
最近6个月的批次创建数量

### 数据来源
`batch` 表的 `create_time` 字段

### SQL查询
```sql
-- 超级管理员
SELECT 
  DATE_FORMAT(create_time, '%Y-%m') as month,
  COUNT(*) as count
FROM batch
WHERE status != -1
  AND create_time >= DATE_SUB(NOW(), INTERVAL 6 MONTH)
GROUP BY DATE_FORMAT(create_time, '%Y-%m')
ORDER BY month ASC;

-- 农户
SELECT 
  DATE_FORMAT(create_time, '%Y-%m') as month,
  COUNT(*) as count
FROM batch
WHERE farmer_id = ?
  AND status != -1
  AND create_time >= DATE_SUB(NOW(), INTERVAL 6 MONTH)
GROUP BY DATE_FORMAT(create_time, '%Y-%m')
ORDER BY month ASC;
```

### 1. 创建 DashboardController

```java
package org.xingchang.xichanginterface.controller;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;
import org.xingchang.xichanginterface.config.ApiResponse;
import org.xingchang.xichanginterface.dto.DashboardStatisticsDTO;
import org.xingchang.xichanginterface.service.DashboardService;

@RestController
@RequestMapping("/api/dashboard")
public class DashboardController {
    
    @Autowired
    private DashboardService dashboardService;
    
    /**
     * 获取驾驶舱统计数据
     */
    @GetMapping("/statistics")
    public ApiResponse<DashboardStatisticsDTO> getStatistics(
        @RequestParam String userId,
        @RequestParam String userRole
    ) {
        DashboardStatisticsDTO statistics = dashboardService.getStatistics(userId, userRole);
        return ApiResponse.success(statistics);
    }
}
```

### 2. 创建 DashboardService

```java
package org.xingchang.xichanginterface.service;

import org.xingchang.xichanginterface.dto.DashboardStatisticsDTO;

public interface DashboardService {
    /**
     * 获取驾驶舱统计数据
     */
    DashboardStatisticsDTO getStatistics(String userId, String userRole);
}
```

### 3. 创建 DashboardServiceImpl

```java
package org.xingchang.xichanginterface.service.impl;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.xingchang.xichanginterface.dto.*;
import org.xingchang.xichanginterface.mapper.DashboardMapper;
import org.xingchang.xichanginterface.service.DashboardService;

import java.util.List;

@Service
public class DashboardServiceImpl implements DashboardService {
    
    @Autowired
    private DashboardMapper dashboardMapper;
    
    @Override
    public DashboardStatisticsDTO getStatistics(String userId, String userRole) {
        DashboardStatisticsDTO statistics = new DashboardStatisticsDTO();
        
        boolean isSuperAdmin = "SUPER_ADMIN".equals(userRole);
        
        // 1. 统计卡片数据
        statistics.setTotalBatches(dashboardMapper.countBatches(userId, isSuperAdmin));
        statistics.setTotalProducts(dashboardMapper.countProducts(userId, isSuperAdmin));
        statistics.setTotalOrders(dashboardMapper.countOrders(userId, isSuperAdmin));
        statistics.setTotalSales(dashboardMapper.sumSales(userId, isSuperAdmin));
        
        // 2. 批次状态分布
        List<ChartDataDTO> batchStatusData = dashboardMapper.getBatchStatusDistribution(userId, isSuperAdmin);
        statistics.setBatchStatusData(batchStatusData);
        
        // 3. 月度批次趋势
        List<TrendDataDTO> batchTrendData = dashboardMapper.getBatchTrend(userId, isSuperAdmin);
        statistics.setBatchTrendData(batchTrendData);
        
        // 4. 订单状态分布
        List<ChartDataDTO> orderStatusData = dashboardMapper.getOrderStatusDistribution(userId, isSuperAdmin);
        statistics.setOrderStatusData(orderStatusData);
        
        // 5. 最近批次列表
        List<RecentBatchDTO> recentBatches = dashboardMapper.getRecentBatches(userId, isSuperAdmin, 5);
        statistics.setRecentBatches(recentBatches);
        
        // 6. 待处理事项
        List<PendingTaskDTO> pendingTasks = dashboardMapper.getPendingTasks(userId, isSuperAdmin);
        statistics.setPendingTasks(pendingTasks);
        
        return statistics;
    }
}
```

    // 统计订单数量
    Long countOrders(@Param("userId") String userId, @Param("isSuperAdmin") boolean isSuperAdmin);
    
    // 统计销售额

    // 批次状态分布
    List<ChartDataDTO> getBatchStatusDistribution(@Param("userId") String userId, @Param("isSuperAdmin") boolean isSuperAdmin);
    
    // 月度批次趋势
    List<TrendDataDTO> getBatchTrend(@Param("userId") String userId, @Param("isSuperAdmin") boolean isSuperAdmin);
    
    // 订单状态分布
    List<ChartDataDTO> getOrderStatusDistribution(@Param("userId") String userId, @Param("isSuperAdmin") boolean isSuperAdmin);
    
    // 最近批次列表
    List<RecentBatchDTO> getRecentBatches(@Param("userId") String userId, @Param("isSuperAdmin") boolean isSuperAdmin, @Param("limit") int limit);
    
    // 待处理事项
    List<PendingTaskDTO> getPendingTasks(@Param("userId") String userId, @Param("isSuperAdmin") boolean isSuperAdmin);
}
