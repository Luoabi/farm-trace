## 三、小程序登录流程

### 方案 A：手机号+密码登录（推荐）

#### 流程图
```
用户打开小程序
    ↓
首次使用：注册账号
    ├─ 输入手机号
    ├─ 输入密码
    ├─ 输入姓名
    └─ 调用 POST /api/register
    ↓
已有账号：登录
    ├─ 输入手机号
    ├─ 输入密码
    └─ 调用 POST /api/login
    ↓
登录成功
    ├─ 保存 token 到本地
    ├─ 保存用户信息到本地
    └─ 跳转到首页
```

#### 小程序代码示例

**1. 注册页面**
```javascript
// pages/register/register.js
Page({
  data: {
    phone: '',
    password: '',
    realName: ''
  },
  
  // 注册
  async handleRegister() {
    const { phone, password, realName } = this.data;
    
    // 验证
    if (!phone || !password || !realName) {
      wx.showToast({ title: '请填写完整信息', icon: 'none' });
      return;
    }
    
    try {
      wx.showLoading({ title: '注册中...' });
      
      const res = await wx.request({
        url: 'https://api.example.com/api/register',
        method: 'POST',
        data: {
          username: phone,
          password: password,
          realName: realName,
          phone: phone,
          role: 'CUSTOMER'
        }
      });
      
      wx.hideLoading();
      
      if (res.data.code === 200) {
        wx.showToast({ title: '注册成功', icon: 'success' });
        
        // 自动登录
        setTimeout(() => {
          wx.redirectTo({ url: '/pages/login/login' });
        }, 1500);
      } else {
        wx.showToast({ 
          title: res.data.message || '注册失败', 
          icon: 'none' 
        });
      }
    } catch (error) {
      wx.hideLoading();
      console.error('注册失败:', error);
      wx.showToast({ title: '注册失败', icon: 'none' });
    }
  }
});
```

**2. 登录页面**
```javascript
// pages/login/login.js
Page({
  data: {
    phone: '',
    password: ''
  },
  
  // 登录
  async handleLogin() {
    const { phone, password } = this.data;
    
    // 验证
    if (!phone || !password) {
      wx.showToast({ title: '请输入手机号和密码', icon: 'none' });
      return;
    }
    
    try {
      wx.showLoading({ title: '登录中...' });
      
      const res = await wx.request({
        url: 'https://api.example.com/api/login',
        method: 'POST',
        data: {
          username: phone,
          password: password
        }
      });
      
      wx.hideLoading();
      
      if (res.data.code === 200) {
        const userData = res.data.data;
        
        // 保存用户信息到本地
        wx.setStorageSync('token', userData.token);
        wx.setStorageSync('userInfo', userData);
        
        wx.showToast({ title: '登录成功', icon: 'success' });
        
        // 跳转到首页
        setTimeout(() => {
          wx.switchTab({ url: '/pages/index/index' });
        }, 1500);
      } else {
        wx.showToast({ 
          title: res.data.message || '登录失败', 
          icon: 'none' 
        });
      }
    } catch (error) {
      wx.hideLoading();
      console.error('登录失败:', error);
      wx.showToast({ title: '登录失败', icon: 'none' });
    }
  },
  
  // 跳转到注册页面
  goToRegister() {
    wx.navigateTo({ url: '/pages/register/register' });
  }
});
```

**3. 请求拦截器（添加 token）**
```javascript
// utils/request.js
const request = (options) => {
  return new Promise((resolve, reject) => {
    // 获取 token
    const token = wx.getStorageSync('token');
    
    wx.request({
      ...options,
      header: {
        'Content-Type': 'application/json',
        'Authorization': token ? `Bearer ${token}` : '',
        ...options.header
      },
      success: (res) => {
        if (res.statusCode === 200) {
          resolve(res.data);
        } else if (res.statusCode === 401) {
          // token 过期，跳转到登录页
          wx.showToast({ title: '请先登录', icon: 'none' });
          wx.redirectTo({ url: '/pages/login/login' });
          reject(res);
        } else {
          reject(res);
        }
      },
      fail: (err) => {
        reject(err);
      }
    });
  });
};

export default request;
```

**4. 使用示例**
```javascript
// pages/order/order.js
import request from '../../utils/request';

Page({
  async loadOrders() {
    try {
      const userInfo = wx.getStorageSync('userInfo');
      
      const res = await request({
        url: 'https://api.example.com/api/order/list-by-customer',
        method: 'POST',
        data: {
          page: 1,
          pageSize: 10
        },
        params: {
          customerId: userInfo.id
        }
      });
      
      console.log('订单列表:', res);
    } catch (error) {
      console.error('获取订单失败:', error);
    }
  }
});
```

## 五、登录状态管理

### 1. 检查登录状态

```javascript
// utils/auth.js
export const checkLogin = () => {
  const token = wx.getStorageSync('token');
  const userInfo = wx.getStorageSync('userInfo');
  
  if (!token || !userInfo) {
    return false;
  }
  
  return true;
};

export const requireLogin = () => {
  if (!checkLogin()) {
    wx.showToast({ title: '请先登录', icon: 'none' });
    wx.redirectTo({ url: '/pages/login/login' });
    return false;
  }
  return true;
};

export const getUserInfo = () => {
  return wx.getStorageSync('userInfo');
};

export const logout = () => {
  wx.removeStorageSync('token');
  wx.removeStorageSync('userInfo');
  wx.redirectTo({ url: '/pages/login/login' });
};
```

### 2. 在需要登录的页面使用

```javascript

    this.loadOrders();
  },
```

## 九、注意事项

### 1. 密码安全

建议在后端实现：
- 密码加密存储（MD5 或 BCrypt）
- 密码强度验证
- 登录失败次数限制

### 2. Token 管理

建议在后端实现：
- JWT Token 生成
- Token 过期时间设置
- Token 刷新机制

### 3. 角色验证

在接口中验证用户角色：
```java
// 只有 CUSTOMER 角色可以创建订单
if (!"CUSTOMER".equals(user.getRole())) {
    throw new RuntimeException("无权限操作");
}
```
